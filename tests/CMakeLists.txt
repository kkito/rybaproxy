find_package(cmocka CONFIG REQUIRED)

function(add_ryba_test name src)
    add_executable(${name} ${src})
    target_link_libraries(${name} PRIVATE cmocka::cmocka rybaproxy_core)
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/include)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

add_ryba_test(test_smoke unit/test_smoke.c)
add_ryba_test(test_config unit/test_config.c)
add_ryba_test(test_router unit/test_router.c)
add_ryba_test(test_auth unit/test_auth.c)
add_ryba_test(test_socks5 unit/test_socks5.c)
add_ryba_test(test_http_proxy unit/test_http_proxy.c)
add_ryba_test(test_stats unit/test_stats.c)
add_ryba_test(test_db unit/test_db.c)
add_ryba_test(test_upstream_probe unit/test_upstream_probe.c)
